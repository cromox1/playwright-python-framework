Example API test

You can use the free API from ReqRes (https://reqres.in) for practice.

tests/api/test_users_api.py

from playwright.sync_api import sync_playwright


def test_get_single_user():

    with sync_playwright() as p:

        request = p.request.new_context()

        response = request.get(
            "https://reqres.in/api/users/2"
        )

        assert response.status == 200

        data = response.json()

        assert data["data"]["id"] == 2

        assert data["data"]["email"] == "janet.weaver@reqres.in"

Run only API tests:

pytest tests/api/

Notice the additional api/ folder.

The idea is similar to the Page Object Model:

pages/ contains page objects for UI.
api/ contains reusable classes for API endpoints.

For example:

# api/users_api.py

class UsersAPI:

    def __init__(self, request):
        self.request = request

    def get_user(self, user_id):
        return self.request.get(
            f"https://reqres.in/api/users/{user_id}"
        )

    def create_user(self, body):
        return self.request.post(
            "https://reqres.in/api/users",
            data=body
        )

Then your test becomes very clean:

from api.users_api import UsersAPI


def test_get_user(api_request):

    users = UsersAPI(api_request)

    response = users.get_user(2)

    assert response.status == 200
