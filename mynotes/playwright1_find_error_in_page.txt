# If you're using Playwright + pytest (Python), there are several ways to detect page errors. The best approach depends on what you mean by "error".


#1) Detect JavaScript errors (Recommended)

Capture all JavaScript errors on the page.

---
import pytest

def test_no_js_errors(page):
    errors = []

    page.on("pageerror", lambda exception: errors.append(str(exception)))

    page.goto("https://example.com")

    assert errors == [], f"JavaScript errors found:\n{errors}"
---

Example output:

AssertionError:
JavaScript errors found:
['ReferenceError: foo is not defined']

#2) Detect Console Errors

Many applications log errors to the browser console instead of throwing exceptions.

---
def test_console_errors(page):
    console_errors = []

    def handle_console(msg):
        if msg.type == "error":
            console_errors.append(msg.text)

    page.on("console", handle_console)

    page.goto("https://example.com")

    assert console_errors == [], f"Console errors:\n{console_errors}"
---

#3) Detect HTTP Errors (404, 500)

Capture failed network responses.

---
def test_http_errors(page):
    failed_requests = []

    def handle_response(response):
        if response.status >= 400:
            failed_requests.append(
                f"{response.status} - {response.url}"
            )

    page.on("response", handle_response)

    page.goto("https://example.com")

    assert failed_requests == [], \
        f"HTTP errors:\n" + "\n".join(failed_requests)
---
Example:

404 - https://example.com/logo.png
500 - https://example.com/api/users

#4) Detect Failed Requests

Sometimes a request never reaches the server.

---
def test_failed_requests(page):
    failures = []

    page.on(
        "requestfailed",
        lambda request: failures.append(
            f"{request.method} {request.url}"
        )
    )

    page.goto("https://example.com")

    assert failures == []
---

#5) Detect Error Message in UI

If your application shows an error banner:

---
from playwright.sync_api import expect

def test_no_error_banner(page):
    page.goto("https://example.com")

    expect(page.locator(".error")).to_have_count(0)
---

or

---
assert not page.locator("text=Something went wrong").is_visible()
---

#6) Check Page Title

Useful if your application has an error page.

---
def test_not_error_page(page):
    page.goto("https://example.com")

    assert page.title() != "Error"
---

#7) Detect "404" or "500" in Page

---
def test_not_404(page):
    page.goto("https://example.com")

    body = page.locator("body").inner_text()

    assert "404" not in body
    assert "500" not in body
    assert "Internal Server Error" not in body
---

#8) Take Screenshot Automatically When Test Fails

In conftest.py:

---
import pytest

@pytest.hookimpl(hookwrapper=True)
def pytest_runtest_makereport(item):
    outcome = yield
    rep = outcome.get_result()

    if rep.when == "call" and rep.failed:
        page = item.funcargs.get("page")
        if page:
            page.screenshot(
                path=f"screenshots/{item.name}.png",
                full_page=True
            )
---

#9) Best Practice: Check Everything Together

You can combine all checks into one fixture so every test automatically monitors for issues.

---
import pytest

@pytest.fixture
def monitor_page(page):
    js_errors = []
    console_errors = []
    http_errors = []

    page.on("pageerror",
            lambda e: js_errors.append(str(e)))

    page.on("console",
            lambda m: console_errors.append(m.text)
            if m.type == "error" else None)

    page.on("response",
            lambda r: http_errors.append(
                f"{r.status} {r.url}"
            ) if r.status >= 400 else None)

    yield page

    assert not js_errors, \
        f"JavaScript errors:\n{js_errors}"

    assert not console_errors, \
        f"Console errors:\n{console_errors}"

    assert not http_errors, \
        f"HTTP errors:\n{http_errors}"
---

Then simply use it:

---
def test_login(monitor_page):
    page = monitor_page

    page.goto("https://example.com/login")

    page.get_by_label("Username").fill("admin")
    page.get_by_label("Password").fill("password")
    page.get_by_role("button", name="Login").click()
---

This is a robust pattern for QA automation because every test automatically fails if it encounters:

✅ JavaScript exceptions
✅ Browser console errors
✅ HTTP 4xx/5xx responses
✅ Failed network requests (if you add the requestfailed listener)

For an enterprise-grade Playwright framework, I recommend implementing this monitoring as an autouse pytest fixture in conftest.py so every test benefits from these checks without needing to include them explicitly.
